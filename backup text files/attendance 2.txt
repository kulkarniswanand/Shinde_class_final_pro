    // Fetch all attendance records for today to calculate overall present/absent counts
    useEffect(() => {

                const today = new Date().toISOString().split('T')[0]; // YYYY-MM-DD format
                
                const response = await fetch('/api/studentAttendance/all');
                if (!response.ok) {
                    throw new Error(`HTTP error! Status: ${response.status}`);
                }
                
                const allAttendance = await response.json();
                
                // Filter for today's attendance across all classes
                const todayAttendance = allAttendance.filter(record => {
                    const recordDate = new Date(record.date).toISOString().split('T')[0];
                    return recordDate === today;
                });
                
                // Count present and absent students
                const presentCount = todayAttendance.filter(r => r.status === 'present').length;
                const absentCount = todayAttendance.filter(r => r.status === 'absent').length;
                
                // Update overall stats
                setOverallStats(prev => {
                    const attendanceRate = prev.totalStudents > 0

                        ...prev,
                        presentCount,
                        absentCount,
                        pendingCount: Math.max(0, prev.totalStudents - presentCount - absentCount),

            const date = new Date();

            const formattedDate = `${date.toLocaleString('default', { month: 'short' })} ${date.getDate()}`;

            // Random attendance rate between 65 and 100
            const rate = Math.floor(Math.random() * 35) + 65;

                date: formattedDate,
                rate: rate

            // Random breakdown data
            const total = 8; // Total students
            const present = Math.floor((rate / 100) * total);
            const absent = total - present;

            breakdownData.push({
                date: formattedDate,
                present: present,

        setAttendanceBreakdownData(breakdownData);

    // Generate attendance rate and breakdown data based on real attendance

            // Group records by date
            const recordsByDate = {};
            attendanceRecords.forEach(record => {
                const recordDate = new Date(record.date);
                const dateStr = recordDate.toISOString().split('T')[0];
                
                if (!recordsByDate[dateStr]) {
                    recordsByDate[dateStr] = { present: 0, absent: 0, date: dateStr };
                }
                
                if (record.status === 'present') {
                    recordsByDate[dateStr].present += 1;
                } else if (record.status === 'absent') {
                    recordsByDate[dateStr].absent += 1;
                }
            });
            
            // Create attendance rate data for charts
            const rateData = Object.values(recordsByDate).map(day => {
                const total = day.present + day.absent;
                const rate = total > 0 ? (day.present / total) * 100 : 0;
                const formattedDate = new Date(day.date).toLocaleDateString('en-US', { month: 'short', day: 'numeric' });

                    date: formattedDate,
                    rate: parseFloat(rate.toFixed(1)),
                    present: day.present,
                    absent: day.absent
                };

            rateData.sort((a, b) => new Date(a.date) - new Date(b.date));

            setAttendanceBreakdownData(rateData);

                    setAttendanceRecords([]);

                setLoadingAttendance(false);

            // Handle both 'studentname' and 'name' field variations

                name: studentName,
                studentClass: studentClass,
                branch: studentBranch,
                status: status

                class: studentClass,

                date: formattedDate,
                status: status

                } else {

                    return [...prev, newRecord];
                }

        } catch (error) {

            alert(`Failed to mark attendance: ${error.message}`);

            date.setDate(startDate.getDate() + i);

            const absentCount = totalStudents - presentCount;

            history.push({
                date: formattedDate,
                dateLabel,
                presentCount,
                absentCount: Math.max(0, absentCount),
                attendanceRate
            });

        return history;

            fetchAttendanceHistory();

            const recordId = record.id ? Number(record.id) : null;
            return recordId === numericId;

            const recordDate = new Date(record.date);
            const recordDateFormatted = recordDate.toISOString().split('T')[0];
            const matchesDate = recordDateFormatted === formattedSelectedDate;
            
            return matchesStudent && matchesDate;

        return record ? record.status : 'pending';

                if (statusFilter === 'absent' && record.absentCount > 0) return true;

                        {/* Search Attendance Records */}

                            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                                {/* Class and Student Filters */}
                                <div>
                                    <label htmlFor="history-class-select" className="block text-sm font-medium text-gray-700 mb-2">
                                        Class
                                    </label>
                                    <select
                                        id="history-class-select"
                                        value={selectedClass}
                                        onChange={(e) => setSelectedClass(Number(e.target.value))}

                                        <option value={0}>All Classes</option>
                                        {classes.map((cls) => (

                                        <div className="absolute inset-y-0 right-0 flex items-center pr-3 pointer-events-none">

                                                                {record.presentCount || 0}

                            </div>
                                    
                                    {/* Pagination */}
                                    {filteredHistory.length > historyRecordsPerPage && (
                                        <div className="mt-4 flex justify-between items-center">
                                            <p className="text-sm text-gray-500">
                                                Showing {Math.min(filteredHistory.length, (historyPage - 1) * historyRecordsPerPage + 1)} to {Math.min(filteredHistory.length, historyPage * historyRecordsPerPage)} of {filteredHistory.length} records
                                            </p>
                                            <div className="flex space-x-2">
                                                <button 
                                                    className="px-3 py-1 bg-gray-200 text-gray-700 rounded text-sm disabled:opacity-50"
                                                    disabled={historyPage === 1}
                                                    onClick={() => setHistoryPage(prev => Math.max(prev - 1, 1))}

                                                    Previous

                                                    className="px-3 py-1 bg-gray-200 text-gray-700 rounded text-sm disabled:opacity-50"
                                                    disabled={historyPage === Math.ceil(filteredHistory.length / historyRecordsPerPage)}
                                                    onClick={() => setHistoryPage(prev => Math.min(prev + 1, Math.ceil(filteredHistory.length / historyRecordsPerPage)))}
                                                >
                                                    Next

                                        <div className="text-center">

                                                {calendarDate.toLocaleString('default', { month: 'long', year: 'numeric' })}
                                            </h3>
                                            <p className="text-sm text-gray-500">
                                                Attendance Overview {selectedClass !== 0 ? `(${classes.find(c => c.id === selectedClass)?.name})` : '(All Classes)'}

                                        {(() => {
                                            const year = calendarDate.getFullYear();
                                            const month = calendarDate.getMonth();
                                            const firstDay = new Date(year, month, 1).getDay();
                                            const daysInMonth = new Date(year, month + 1, 0).getDate();
                                            
                                            // Calendar data for quick lookup
                                            const calendarData = getCalendarData();
                                            const dateMap = {};
                                            calendarData.forEach(item => {
                                                const date = new Date(item.date);
                                                if (date.getMonth() === month && date.getFullYear() === year) {
                                                    dateMap[date.getDate()] = item;

                                            const days = [];

                                            for (let i = 0; i < firstDay; i++) {

                                                    <div key={`empty-${i}`} className="p-2 border rounded bg-gray-50"></div>

                                            }
                                            
                                            // Days of the month
                                            for (let day = 1; day <= daysInMonth; day++) {
                                                const today = new Date();
                                                const isToday = day === today.getDate() && 
                                                               month === today.getMonth() && 
                                                               year === today.getFullYear();
                                                
                                                const dateData = dateMap[day];

                                                // Prepare formatted date for view details

                                                const detailsData = {
                                                    dateLabel: formattedDay.toLocaleDateString('en-US', {

                                                        day: 'numeric'
                                                    }),
                                                    date: formattedDay.toISOString().split('T')[0]
                                                };
                                                
                                                if (hasAttendance) {
                                                    detailsData.presentCount = dateData.present || 0;
                                                    detailsData.absentCount = dateData.absent || 0;
                                                    detailsData.attendanceRate = dateData.rate || 0;
                                                }
                                                
                                                days.push(
                                                    <div 
                                                        key={`day-${day}`} 
                                                        className={`p-2 border rounded 
                                                            ${isToday ? 'bg-blue-50 border-blue-200' : ''}
                                                            ${hasAttendance ? 'cursor-pointer hover:bg-gray-100' : 'bg-white'}
                                                        `}
                                                        onClick={hasAttendance ? () => viewRecordDetails(detailsData) : undefined}
                                                    >
                                                        <div className="flex justify-between items-center">
                                                            <span className={`${isToday ? 'font-bold text-blue-600' : 'text-gray-700'}`}>{day}</span>
                                                            {hasAttendance && (
                                                                <span className={`text-xs px-1.5 py-0.5 rounded-full ${
                                                                    dateData.rate >= 90 
                                                                        ? 'bg-green-100 text-green-800' 
                                                                        : dateData.rate >= 75
                                                                        ? 'bg-yellow-100 text-yellow-800'
                                                                        : 'bg-red-100 text-red-800'
                                                                }`}>
                                                                    {dateData.rate}%
                                                                </span>
                                                            )}
                                                        </div>
                                                        {hasAttendance && (
                                                            <div className="mt-1 text-xs">
                                                                <span className="text-green-600">P: {dateData.present}</span>
                                                                <span className="text-red-600 ml-1">A: {dateData.absent}</span>
                                                            </div>
                                                        )}
                                                    </div>
                                                );
                                            }
                                            
                                            return days;
                                        })()}
                                    </div>
                                    
                                    {/* Calendar legend */}
                                    <div className="mt-6 flex flex-wrap justify-center gap-4 text-sm text-gray-500">
                                        <div className="flex items-center">
                                            <div className="h-3 w-3 rounded-full bg-green-100 mr-1"></div>
                                            <span>≥ 90% (Excellent)</span>
                                        </div>
                                        <div className="flex items-center">
                                            <div className="h-3 w-3 rounded-full bg-yellow-100 mr-1"></div>
                                            <span>75-90% (Good)</span>
                                        </div>
                                        <div className="flex items-center">
                                            <div className="h-3 w-3 rounded-full bg-red-100 mr-1"></div>
                                            <span>&lt; 75% (Needs Improvement)</span>
                                        </div>
                                        <div className="flex items-center">
                                            <div className="h-3 w-3 rounded-full bg-blue-100 mr-1"></div>
                                            <span>Today</span>
                                        </div>
                                    </div>
                                </div>
                            )}
                            
                            {/* Record Details Modal */}
                            {showRecordDetails && selectedHistoryRecord && (
                                <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50">
                                    <div className="bg-white rounded-lg shadow-lg p-6 max-w-2xl w-full mx-4">
                                        <div className="flex justify-between items-start mb-4">
                                            <h3 className="text-xl font-bold text-gray-900">
                                                Attendance Details: {selectedHistoryRecord.dateLabel}
                                            </h3>
                                            <button 
                                                className="text-gray-500 hover:text-gray-700"
                                                onClick={() => setShowRecordDetails(false)}
                                            >
                                                <svg xmlns="http://www.w3.org/2000/svg" className="h-6 w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                                                    <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
                                                </svg>
                                            </button>
                                        </div>
                                        
                                        <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-6">
                                            <div className="bg-green-50 p-4 rounded">
                                                <p className="text-sm text-green-700 font-medium">Present</p>
                                                <p className="text-2xl font-bold text-green-700">{recordDetails.presentStudents.length || 0}</p>
                                            </div>
                                            <div className="bg-red-50 p-4 rounded">
                                                <p className="text-sm text-red-700 font-medium">Absent</p>
                                                <p className="text-2xl font-bold text-red-700">{recordDetails.absentStudents.filter(s => !s.missing).length || 0}</p>
                                            </div>
                                            <div className="bg-blue-50 p-4 rounded">
                                                <p className="text-sm text-blue-700 font-medium">Attendance Rate</p>
                                                <p className="text-2xl font-bold text-blue-700">
                                                    {recordDetails.attendanceRate || 0}%
                                                </p>
                                            </div>
                                        </div>
                                        
                                        <div className="mb-6">
                                            <h4 className="text-lg font-medium text-gray-800 mb-2">Students Present ({recordDetails.presentStudents.length})</h4>
                                            <div className="max-h-40 overflow-y-auto p-2 bg-gray-50 rounded">
                                                {recordDetails.loading ? (
                                                    <div className="flex justify-center items-center p-4">
                                                        <div className="animate-spin rounded-full h-6 w-6 border-t-2 border-b-2 border-blue-500"></div>
                                                        <span className="ml-2 text-sm text-blue-500">Loading students...</span>
                                                    </div>
                                                ) : recordDetails.presentStudents.length > 0 ? (
                                                    <ul className="divide-y divide-gray-200">
                                                        {recordDetails.presentStudents.map((student, index) => (
                                                            <li key={`present-${student.id}-${index}`} className="py-2 flex items-center">
                                                                <div className="h-8 w-8 rounded-full bg-green-100 flex items-center justify-center text-green-600 font-bold text-sm mr-3">
                                                                    {student.name ? student.name.charAt(0) : 'S'}
                                                                </div>
                                                                <div>
                                                                    <p className="text-sm font-medium text-gray-700">{student.name}</p>
                                                                    <p className="text-xs text-gray-500">
                                                                        ID: {student.id} • Class: {student.class} • Branch: {student.branch || 'N/A'}
                                                                    </p>
                                                                </div>
                                                            </li>
                                                        ))}
                                                    </ul>
                                                ) : recordDetails.error ? (
                                                    <p className="text-red-500 italic text-sm p-2">
                                                        Error loading data: {recordDetails.error}
                                                    </p>
                                                ) : (
                                                    <p className="text-gray-500 italic text-sm p-2">
                                                        No students marked present on this date.
                                                    </p>
                                                )}
                                            </div>
                                        </div>
                                        
                                        <div className="mb-6">
                                            <h4 className="text-lg font-medium text-gray-800 mb-2">Students Absent ({recordDetails.absentStudents.length})</h4>
                                            <div className="max-h-40 overflow-y-auto p-2 bg-gray-50 rounded">
                                                {recordDetails.loading ? (
                                                    <div className="flex justify-center items-center p-4">
                                                        <div className="animate-spin rounded-full h-6 w-6 border-t-2 border-b-2 border-blue-500"></div>
                                                        <span className="ml-2 text-sm text-blue-500">Loading students...</span>
                                                    </div>
                                                ) : recordDetails.absentStudents.length > 0 ? (
                                                    <ul className="divide-y divide-gray-200">
                                                        {recordDetails.absentStudents.map((student, index) => (
                                                            <li key={`absent-${student.id}-${index}`} className="py-2 flex items-center">
                                                                <div className="h-8 w-8 rounded-full bg-red-100 flex items-center justify-center text-red-600 font-bold text-sm mr-3">
                                                                    {student.name ? student.name.charAt(0) : 'S'}
                                                                </div>
                                                                <div>
                                                                    <p className="text-sm font-medium text-gray-700">{student.name}</p>
                                                                    <p className="text-xs text-gray-500">
                                                                        ID: {student.id} • Class: {student.class} • Branch: {student.branch || 'N/A'}
                                                                    </p>
                                                                </div>
                                                            </li>
                                                        ))}
                                                    </ul>
                                                ) : recordDetails.error ? (
                                                    <p className="text-red-500 italic text-sm p-2">
                                                        Error loading data: {recordDetails.error}
                                                    </p>
                                                ) : (
                                                    <p className="text-gray-500 italic text-sm p-2">
                                                        No students marked absent on this date.
                                                    </p>
                                                )}
                                            </div>
                                        </div>
                                        
                                        <div className="flex justify-end space-x-3 mt-6">
                                            <CSVLink 
                                                data={getCSVData()}
                                                filename={`attendance_report_${selectedHistoryRecord.dateLabel || 'current'}_${selectedHistoryRecord.className || 'all'}.csv`}
                                                className="bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600"
                                                target="_blank"
                                            >
                                                Export CSV
                                            </CSVLink>
                                            <button
                                                className="bg-green-500 text-white px-4 py-2 rounded hover:bg-green-600"
                                                onClick={handleExportFromModal}
                                            >
                                                Export PDF
                                            </button>
                                            <button
                                                className="bg-gray-200 text-gray-700 px-4 py-2 rounded hover:bg-gray-300"
                                                onClick={() => setShowRecordDetails(false)}
                                            >
                                                Close
                                            </button>
                                        </div>
                                    </div>
                                </div>
                            )}
                        </div>
                    </>
                )}
            </div>

            {/* Footer */}
            <footer className="w-full bg-gray-800 text-white py-4 text-center mt-auto">
                <p className="text-sm">© 2025 Attendance Management System</p>
            </footer>
        </div>
    );
};

export default AttendanceDashboard;
