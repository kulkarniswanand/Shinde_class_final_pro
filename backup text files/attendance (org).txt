const AttendanceDashboard = () => {
    // State management

    const [attendanceRecords, setAttendanceRecords] = useState([]);
    const [classes, setClasses] = useState([]);
    const [selectedClass, setSelectedClass] = useState(0);
    const [selectedDate, setSelectedDate] = useState("2025-04-04");
    const [searchTerm, setSearchTerm] = useState("");
    const [loading, setLoading] = useState(true);
    const [stats, setStats] = useState({
        totalStudents: 0,
        presentCount: 0,
        absentCount: 0,
        pendingCount: 0
    });
    const [currentPage, setCurrentPage] = useState(1);
    const [recordsPerPage] = useState(10);
    const [attendanceHistory, setAttendanceHistory] = useState([]);
    const [attendanceRateData, setAttendanceRateData] = useState([]);
    const [attendanceBreakdownData, setAttendanceBreakdownData] = useState([]);
    const [activeView, setActiveView] = useState('dashboard'); // Options: 'dashboard', 'roster', 'history'
    const [historyDateRange, setHistoryDateRange] = useState({
        startDate: "2025-03-28",
        endDate: "2025-04-04"
    });
    const [selectedStudent, setSelectedStudent] = useState(0);

    // Fetch initial data
    useEffect(() => {
        const fetchData = async () => {
            try {
                setLoading(true);

                // In a real implementation, these would be API calls to your MySQL database
                const studentsResponse = await axios.get('/api/students');
                const classesResponse = await axios.get('/api/classes');

                setStudents(studentsResponse.data);
                setClasses(classesResponse.data);

                if (classesResponse.data.length > 0) {

                    { id: 4, name: "Sarah Williams", rollNumber: "CS004", profileImage: "https://via.placeholder.com/150" },
                    { id: 5, name: "Michael Brown", rollNumber: "CS005", profileImage: "https://via.placeholder.com/150" },
                    { id: 6, name: "Emma Davis", rollNumber: "CS006", profileImage: "https://via.placeholder.com/150" },
                    { id: 7, name: "David Wilson", rollNumber: "CS007", profileImage: "https://via.placeholder.com/150" },
                    { id: 8, name: "Olivia Martinez", rollNumber: "CS008", profileImage: "https://via.placeholder.com/150" },
                ]);

                setClasses([
                    { id: 1, name: "Introduction to Programming", courseCode: "CS101" },
                    { id: 2, name: "Data Structures", courseCode: "CS102" },
                    { id: 3, name: "Algorithms", courseCode: "CS103" },
                ]);

                setSelectedClass(1);
            } finally {
                setLoading(false);
            }
        };

        fetchData();
        generateMockHistoryData();

        // Create initial mock attendance records to match the image
        setTimeout(() => {
            if (students.length > 0) {
                const mockInitialRecords = [
                    {
                        id: 1,
                        studentId: 1, // John Smith
                        date: selectedDate,
                        status: 'absent',
                        remarks: "Medical appointment"
                    },
                    {
                        id: 2,
                        studentId: 2, // Emily Johnson
                        date: selectedDate,
                        status: 'present',
                        remarks: ""
                    }
                ];

                setAttendanceRecords(mockInitialRecords);

                // Update stats for the mock data
                const present = mockInitialRecords.filter(r => r.status === 'present').length;
                const absent = mockInitialRecords.filter(r => r.status === 'absent').length;
                const pending = students.length - present - absent;

                setStats({
                    totalStudents: students.length,
                    presentCount: present,
                    absentCount: absent,
                    pendingCount: pending

        // Generate attendance rate data for 7 days

        const breakdownData = [];
        const now = new Date();

        for (let i = 6; i >= 0; i--) {
            const date = new Date();
            date.setDate(now.getDate() - i);
            const formattedDate = `${date.toLocaleString('default', { month: 'short' })} ${date.getDate()}`;

            // Random attendance rate between 65 and 100
            const rate = Math.floor(Math.random() * 35) + 65;

            rateData.push({
                date: formattedDate,
                rate: rate
            });

            // Random breakdown data
            const total = 8; // Total students
            const present = Math.floor((rate / 100) * total);
            const absent = total - present;

            breakdownData.push({
                date: formattedDate,
                present: present,
                absent: absent
            });

        setAttendanceRateData(rateData);
        setAttendanceBreakdownData(breakdownData);

        const fetchAttendance = async () => {
            if (selectedClass === 0) return;

            try {
                setLoading(true);

                // In a real implementation, this would be an API call to your MySQL database
                const response = await axios.get(`/api/attendance?classId=${selectedClass}&date=${selectedDate}`);
                setAttendanceRecords(response.data);

                // Calculate stats
                const present = response.data.filter(r => r.status === 'present').length;
                const absent = response.data.filter(r => r.status === 'absent').length;
                const pending = students.length - present - absent;

                setStats({
                    totalStudents: students.length,
                    presentCount: present,
                    absentCount: absent,
                    pendingCount: pending
                });

            } catch (error) {
                console.error("Failed to fetch attendance records:", error);

                const mockRecords = students.map(student => ({
                    id: student.id,
                    studentId: student.id,
                    date: selectedDate,
                    status: statuses[Math.floor(Math.random() * 2)],
                    remarks: "",
                    student: student

                // Calculate stats from mock data

                const absent = mockRecords.filter(r => r.status === 'absent').length;
                const pending = students.length - present - absent;

                    totalStudents: students.length,
                    presentCount: present,
                    absentCount: absent,
                    pendingCount: pending

                setLoading(false);

            fetchAttendance();

        if (selectedClass === 0) return;

        try {
            // In a real implementation, this would be an API call to your MySQL database
            const response = await axios.post('/api/attendance', {
                studentId,
                classId: selectedClass,
                date: selectedDate,
                status

            setAttendanceRecords(prev => {
                const index = prev.findIndex(r => r.studentId === studentId);
                if (index >= 0) {
                    // Update existing record
                    const updated = [...prev];
                    updated[index] = { ...updated[index], status };
                    return updated;
                } else {
                    // Add new record
                    const student = students.find(s => s.id === studentId);
                    return [...prev, {
                        id: Date.now(), // Temporary ID
                        studentId,
                        date: selectedDate,
                        status,
                        remarks: "",
                        student

            const recordIndex = updatedRecords.findIndex(r => r.studentId === studentId);
            if (recordIndex >= 0) {
                updatedRecords[recordIndex].status = status;

                updatedRecords.push({
                    studentId,
                    status,
                    date: selectedDate
                });

            const absent = updatedRecords.filter(r => r.status === 'absent').length;

                totalStudents: students.length,
                presentCount: present,
                absentCount: absent,
                pendingCount: pending

            console.error("Failed to mark attendance:", error);

        const daysDiff = Math.floor((endDate - startDate) / (1000 * 60 * 60 * 24)) + 1;

        for (let i = 0; i < daysDiff; i++) {
            const date = new Date(startDate);
            date.setDate(startDate.getDate() + i);

            // Skip weekends if desired
            if (date.getDay() === 0 || date.getDay() === 6) continue;

            const formattedDate = date.toISOString().split('T')[0];
            const dateLabel = date.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });

            // Random data for demonstration
            const totalStudents = 8;
            const attendanceRate = Math.floor(Math.random() * 35) + 65;
            const presentCount = Math.floor((attendanceRate / 100) * totalStudents);
            const absentCount = totalStudents - presentCount;

            history.push({
                date: formattedDate,
                dateLabel,
                presentCount,
                absentCount: Math.max(0, absentCount),
                attendanceRate
            });
        }

        return history;
    };

        if (activeView === 'history') {
            const historyData = generateAttendanceHistory();
            setAttendanceHistory(historyData);
        }

            <div className="flex justify-center items-center h-64">

                return "bg-red-100 text-red-800";

                return "bg-gray-100 text-gray-800";

        <div className="min-h-screen bg-gray-100 flex flex-col">

                                    : 'text-gray-600'

                            <span className="text-current">
                                <Icons.CalendarIcon />

                            <span>Dashboard</span>
                        </button>

                            className={`px-4 py-2 flex items-center space-x-2 ${activeView === 'roster'
                                    ? 'text-blue-600'
                                    : 'text-gray-600'
                                }`}
                            onClick={() => setActiveView('roster')}

                            <span className="text-current">
                                <Icons.UserIcon />
                            </span>
                            <span>Class Roster</span>
                        </button>

                            className={`px-4 py-2 flex items-center space-x-2 ${activeView === 'history'
                                    ? 'text-blue-600'
                                    : 'text-gray-600'
                                }`}
                            onClick={() => setActiveView('history')}
                        >
                            <span className="text-current">
                                <Icons.CheckIcon />
                            </span>
                            <span>Attendance History</span>
                        </button>
                    </div>
                </div>
            </header>

            <div className="container mx-auto px-8 py-8 flex-1">
                {activeView === 'dashboard' && (
                    <>
                        <h2 className="text-2xl font-bold text-gray-800 mb-8">Attendance Dashboard</h2>

                            {/* Total Students Card */}
                            <div className="bg-white rounded-lg shadow p-6">
                                <div className="flex justify-between items-start">
                                    <div>
                                        <p className="text-sm font-medium text-gray-500">Total Students</p>
                                        <p className="text-3xl font-bold mt-1">{stats.totalStudents || 8}</p>
                                    </div>
                                    <div className="p-3 bg-blue-100 rounded-full">
                                        <span className="text-blue-600">
                                            <Icons.UserIcon />
                                        </span>
                                    </div>
                                </div>
                            </div>

                            {/* Present Today Card */}
                            <div className="bg-white rounded-lg shadow p-6">
                                <div className="flex justify-between items-start">
                                    <div>
                                        <p className="text-sm font-medium text-gray-500">Present Today</p>
                                        <p className="text-3xl font-bold mt-1">{stats.presentCount || 5}</p>
                                        <p className="text-xs text-green-600 mt-1">+2.5% from last week</p>

                                            <Icons.CheckIcon />
                                        </span>
                                    </div>
                                </div>
                            </div>

                            <div className="bg-white rounded-lg shadow p-6">
                                <div className="flex justify-between items-start">
                                    <div>
                                        <p className="text-sm font-medium text-gray-500">Absent Today</p>
                                        <p className="text-3xl font-bold mt-1">{stats.absentCount || 1}</p>

                                    </div>

                                        <span className="text-red-600">
                                            <Icons.XMarkIcon />
                                        </span>
                                    </div>
                                </div>

                                        <CartesianGrid strokeDasharray="3 3" vertical={false} />
                                        <XAxis
                                            dataKey="date"

                                            tickLine={false}

                                            domain={[0, 100]}
                                            axisLine={false}
                                            tickLine={false}
                                            ticks={[0, 25, 50, 75, 100]}
                                        />
                                        <Tooltip />
                                        <Legend />
                                        <Line
                                            type="monotone"
                                            dataKey="rate"
                                            stroke="#3b82f6"
                                            activeDot={{ r: 8 }}
                                            name="Attendance Rate"
                                            dot={{ stroke: '#3b82f6', strokeWidth: 2, r: 4 }}

                                </ResponsiveContainer>
                            </div>
                        </div>

                        {/* Daily Attendance Breakdown */}
                        <div className="bg-white rounded-lg shadow p-8 mb-10">
                            <h3 className="text-lg font-medium text-gray-800 mb-6">Daily Attendance Breakdown</h3>
                            <div className="h-72">
                                <ResponsiveContainer width="100%" height="100%">
                                    <BarChart data={attendanceBreakdownData}>
                                        <CartesianGrid strokeDasharray="3 3" vertical={false} />

                                            <td className="px-6 py-4 whitespace-nowrap text-sm font-medium text-gray-900">Fri, Apr 4</td>
