import React, { useState } from "react";
import { motion } from "framer-motion";

const StaffRegistrationForm = () => {
  const [formData, setFormData] = useState({
    name: "",
    contact: "",
    designation: "",
    join_date: new Date().toISOString().split("T")[0],
    email: "",
    address: "",
    branch: "",
  });

  const [errors, setErrors] = useState({});
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [serverMessage, setServerMessage] = useState("");

  const handleInputChange = (e) => {
    const { name, value } = e.target;
    setFormData({
      ...formData,
      [name]: value,
    });
  };

  const validateForm = () => {
    const newErrors = {};
    if (!formData.name.trim()) newErrors.name = "Name is required.";
    if (!formData.contact.match(/^\d{10}$/)) newErrors.contact = "Contact must be 10 digits.";
    if (!formData.designation.trim()) newErrors.designation = "Designation is required.";
    if (!formData.email.match(/^\S+@\S+\.\S+$/)) newErrors.email = "Invalid email address.";
    if (!formData.address.trim()) newErrors.address = "Address is required.";

    // Case-insensitive branch validation
    const validBranches = ["pune", "chinchani", "palus"];
    if (!validBranches.includes(formData.branch.trim().toLowerCase())) {
      newErrors.branch = "Branch must be Pune, Chinchani, or Palus (case-insensitive).";
    }

    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!validateForm()) return;

    setIsSubmitting(true);
    setServerMessage("");

    try {
      const response = await fetch(`${process.env.REACT_APP_BACKEND_URL}/api/staff`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(formData),
      });

      if (!response.ok) {
        const errorData = await response.json();
        throw new Error(errorData.message || "Failed to register staff");
      }

      setServerMessage("Staff registered successfully!");
      setFormData({
        name: "",
        contact: "",
        designation: "",
        join_date: new Date().toISOString().split("T")[0],
        email: "",
        address: "",
        branch: "",
      });
      setErrors({});
    } catch (error) {
      console.error("Error registering staff:", error);
      setServerMessage(error.message || "Failed to register staff. Please try again.");
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-gray-800 to-gray-900 text-white px-6 py-10">
      <motion.div
        initial={{ opacity: 0, y: -20 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.5 }}
        className="bg-gray-700 p-10 rounded-2xl shadow-xl w-full max-w-5xl"
      >
        <h1 className="text-3xl font-extrabold text-center mb-6 text-yellow-400">
          Staff Registration Form
        </h1>

        {serverMessage && (
          <p className={`text-center mb-4 ${serverMessage.includes("successfully") ? "text-green-400" : "text-red-400"}`}>
            {serverMessage}
          </p>
        )}

        <form onSubmit={handleSubmit} className="grid grid-cols-2 gap-x-8 gap-y-6">
          {[
            { name: "name", type: "text", label: "Name" },
            { name: "contact", type: "text", label: "Contact" },
            { name: "designation", type: "text", label: "Designation" },
            { name: "join_date", type: "date", label: "Join Date" },
            { name: "email", type: "email", label: "Email" },
            { name: "address", type: "text", label: "Address" },
            { name: "branch", type: "text", label: "Branch" },
          ].map((field, index) => (
            <motion.div
              key={field.name}
              initial={{ opacity: 0, x: index % 2 === 0 ? -20 : 20 }}
              animate={{ opacity: 1, x: 0 }}
              transition={{ duration: 0.3, delay: 0.1 * index }}
              className="flex flex-col"
            >
              <label className="block text-sm font-medium capitalize text-yellow-300 mb-2">
                {field.label}
              </label>
              <input
                type={field.type}
                name={field.name}
                value={formData[field.name]}
                onChange={handleInputChange}
                className="w-full px-4 py-3 bg-gray-600 border border-gray-500 rounded-lg focus:ring-2 focus:ring-yellow-400"
                required
              />
              {errors[field.name] && <p className="text-red-400 text-sm mt-1">{errors[field.name]}</p>}
            </motion.div>
          ))}

          <button
            type="submit"
            className="col-span-2 w-full bg-yellow-500 hover:bg-yellow-600 text-gray-900 py-3 px-4 rounded-lg font-semibold transition-all duration-300"
            disabled={isSubmitting}
          >
            Submit
          </button>
        </form>
      </motion.div> 
    </div>
  );
};

export default StaffRegistrationForm;
